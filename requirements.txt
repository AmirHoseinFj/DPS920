# Simulator networking
# The Udacity simulator is a Unity build using websocket-sharp. It opens a
# websocket directly and never completes the namespace handshake that
# python-socketio 5.x requires. Against a 5.x server the connection is
# established and telemetry arrives, but every event is then dropped with
# "None is not connected to namespace /", so the car never moves - which looks
# exactly like a too-old version and is very easy to misdiagnose.
python-socketio==4.6.1
python-engineio==3.13.2
eventlet==0.33.3
# Flask 2.1.3 and friends are pinned because eventlet 0.33.3 predates the
# Werkzeug 2.1+ / itsdangerous 2.1+ API changes.
Flask==2.1.3
Werkzeug==2.0.3
itsdangerous==2.0.1
Jinja2==3.0.3
MarkupSafe==2.0.1
greenlet==2.0.2
dnspython==2.2.1
